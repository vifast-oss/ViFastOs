// KT-241 A3 — đối chứng dương: fn-pointer/generic/trait-object với `NonReadyTarget` ⇒ XANH.
use crate::{KError, Machine, NonReadyTarget, ObjId};

pub fn positive(m: &mut Machine, t: ObjId) {
    let f = Machine::transition_tcb;
    let _ = f(m, t, NonReadyTarget::Inactive);
    let g: fn(&mut Machine, ObjId, NonReadyTarget) -> Result<(), KError> = Machine::transition_tcb;
    let _ = g(m, t, NonReadyTarget::Inactive);
    let h: &dyn Fn(&mut Machine, ObjId, NonReadyTarget) -> Result<(), KError> =
        &Machine::transition_tcb;
    let _ = h(m, t, NonReadyTarget::Inactive);
}
