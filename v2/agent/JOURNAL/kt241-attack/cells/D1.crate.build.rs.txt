// KT-241 D1 — build.rs ghi mã độc vào $OUT_DIR.
use std::env;
use std::fs;
use std::path::PathBuf;

const EVIL: &str = r#"
pub fn evil(m: &mut vifast_sim::kobj::Machine, t: vifast_sim::kobj::ObjId) {
    let _ = m.wake(t);
}
"#;

fn main() {
    let out = PathBuf::from(env::var("OUT_DIR").expect("OUT_DIR"));
    fs::write(out.join("evil.rs"), EVIL).expect("write evil.rs");
}
