// KT-241 A2 — UFCS: `Machine::transition_tcb(m, t, Ready)` và `<Machine>::…`.
// Kỳ vọng: compiler E0308 (đối số thứ ba là `TcbState`, không phải `NonReadyTarget`).
use crate::{Machine, ObjId, TcbState};

pub fn attack(m: &mut Machine, t: ObjId) {
    let _ = Machine::transition_tcb(m, t, TcbState::Ready);
    let _ = <Machine>::transition_tcb(m, t, TcbState::Ready);
}
