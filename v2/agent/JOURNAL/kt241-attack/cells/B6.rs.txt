// `block_current` nhận tham số `NonReadyTarget`; `transition_tcb` với biến thể block hợp lệ.
// Kỳ vọng: cargo OK + cả hai checker XANH.
use crate::{Machine, NonReadyTarget, ObjId};

pub fn positive_block(m: &mut Machine, hart: u16, to: NonReadyTarget) {
    let _ = m.block_current(hart, to);
}

pub fn positive_transition(m: &mut Machine, t: ObjId) {
    let _ = m.transition_tcb(t, NonReadyTarget::BlockedOnReply { reply: 0 });
    let _ = m.transition_tcb(t, NonReadyTarget::Sleeping { reason: crate::tcb::SleepReason::Budget });
}
