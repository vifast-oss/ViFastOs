// KT-241 E1 — đường vòng qua chính ranh giới mới: `impl From<TcbState> for NonReadyTarget`.
// API-SPEC §1 cấm mọi `From`/`TryFrom` công khai từ `TcbState` sang `NonReadyTarget`.
// Kỳ vọng: cargo OK (impl hợp lệ về orphan rule); lớp 1 PHẢI ĐỎ (luật mới của C);
// lớp 2 ĐỎ nếu C thêm luật — nếu XANH ghi nhận khoảng trống lớp 2.
use crate::tcb::TcbState;
use crate::NonReadyTarget;

impl From<TcbState> for NonReadyTarget {
    fn from(s: TcbState) -> Self {
        match s {
            TcbState::Ready => NonReadyTarget::Running { hart: 0 },
            _ => NonReadyTarget::Inactive,
        }
    }
}

pub fn attack(m: &mut crate::Machine, t: crate::ObjId) {
    let _ = m.transition_tcb(t, TcbState::Ready.into());
}
