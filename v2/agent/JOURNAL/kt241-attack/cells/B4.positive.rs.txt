// KT-241 B4 — đối chứng dương: `const`/`static` kiểu `NonReadyTarget` ⇒ XANH.
use crate::{Machine, NonReadyTarget, ObjId};

const R: NonReadyTarget = NonReadyTarget::Inactive;
static S: NonReadyTarget = NonReadyTarget::Stopping;

pub fn positive(m: &mut Machine, t: ObjId) {
    let _ = m.transition_tcb(t, R);
    let _ = m.transition_tcb(t, S);
}
