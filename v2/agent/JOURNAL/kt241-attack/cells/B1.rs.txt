// KT-241 B1 — helper CÙNG TỆP nhận `TcbState` rồi gọi `transition_tcb`.
// Kỳ vọng: compiler E0308 tại thân helper (không phải tại call site).
use crate::{Machine, ObjId, TcbState};

fn helper(m: &mut Machine, t: ObjId, s: TcbState) {
    let _ = m.transition_tcb(t, s);
}

pub fn attack(m: &mut Machine, t: ObjId) {
    helper(m, t, TcbState::Ready);
}
